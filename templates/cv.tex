% A list whose \item[label] hangs in the left margin, right-aligned
% against the text block.
\newenvironment{dated}{%
  \list{}{%
    \setlength{\leftmargin}{0pt}%
    \setlength{\labelwidth}{\gutter}%
    \setlength{\labelsep}{0pt}%
    \setlength{\itemindent}{0pt}%
    \setlength{\topsep}{0pt}%

    \setlength{\parsep}{0pt}%
    \setlength{\itemsep}{0.55em}%
    \renewcommand{\makelabel}[1]{\hfill{\color{muted}##1}\hspace{1em}}}}%
  {\endlist}

    \setlength{\itemsep}{0.35em}%
    \renewcommand{\makelabel}[1]{\smash{\parbox[t]{\labelwidth}{\RaggedRight\hyphenpenalty=10000\bfseries ##1}}\hfill}}}%
